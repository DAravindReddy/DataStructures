Data structure - stack:
=======================

first in last out(FILO)
stack-operations:
------------------
1. push operation:
->intial top = 0 as an index in an array.
->stack is full when size == top + 1. whenever we are pushing the data to the stack top will increment.
-> push operation fails whenever top == size - 1

2. pop operation:
->to delete an element we are using pop in stack.
->it is not possible to delete an element in array, but in stack when you want to delete from stack just decrement the top.
->pop operation fails whenever stack empty. when top == -1

3. peek operation:
->used to print top most element of the stack.

Create Stack:
-------------
typedef struct
{
	unsigned int capacity; //size
	int top;
	int *item;// array holding the stack elements
}stack_t;


//function to create a stack
int create_stack(stack_t stack, int size)
{
	stack->item = malloc(size*sizeof(int));
	if(stack->item == NULL)
	{
		return e_false;
	}
	stack->capacity = size;
	stack->top = -1;
	return e_true;
}
//time complaxity of push operation O(1)
int push(stack_t stack, int element)
{
	if((stack->top) == capacity - 1)
		return e_false;
	stack->top = stack->top + 1;
	stack->item[stack->top] = element;
	return e_true;
}

//time complecity of pop operation O(1)
int pop(stack_t stack, int element)
{
	if(stack->top == -1)
		return e_false;
	element = stack->item[stack->top];
	stack->top = stack->top - 1;
	return e_true;
}

int peek(stack_t stack)
{
	if(stack->top == -1)
		return e_false;
	return stack->item[stack->top];
}
	

int destroy_stack(stack_t stack)
{
	stack->capacity = 0;
	stack->top = -1;
	stack->item = NULL;
}
